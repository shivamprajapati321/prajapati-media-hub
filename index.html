Supabase SQL Starter

create table users (
  id uuid primary key default gen_random_uuid(),
  name text,
  phone text,
  role text,
  status text default 'active',
  created_at timestamp default now()
);

create table orders (
  id uuid primary key default gen_random_uuid(),
  order_no text,
  company text,
  person text,
  contact text,
  media text,
  total_qty numeric,
  final_amount numeric,
  status text,
  created_at timestamp default now()
);

create table order_locations (
  id uuid primary key default gen_random_uuid(),
  order_id uuid references orders(id),
  location text,
  qty numeric,
  design_type text,
  design_file text
);

create table design_tasks (
  id uuid primary key default gen_random_uuid(),
  order_id uuid references orders(id),
  designer text,
  status text,
  client_approval text
);

create table printing_jobs (
  id uuid primary key default gen_random_uuid(),
  order_id uuid references orders(id),
  operator text,
  machine text,
  total_qty numeric,
  printed_qty numeric,
  ink_used text,
  roll_used text,
  status text
);

create table stitching_entries (
  id uuid primary key default gen_random_uuid(),
  order_id uuid references orders(id),
  worker text,
  qty numeric,
  payment_type text,
  amount numeric,
  ot_amount numeric
);

create table dispatch_entries (
  id uuid primary key default gen_random_uuid(),
  order_id uuid references orders(id),
  dispatch_type text,
  team_or_party text,
  bundles numeric,
  pcs numeric,
  vendor text,
  lr_number text
);

create table execution_photos (
  id uuid primary key default gen_random_uuid(),
  order_id uuid references orders(id),
  vehicle_no text,
  gps text,
  photo_url text,
  ocr_status text
);

create table accounts_entries (
  id uuid primary key default gen_random_uuid(),
  order_id uuid references orders(id),
  type text,
  amount numeric,
  mode text,
  status text
);
